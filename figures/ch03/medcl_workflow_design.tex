\node[choice,below=7mm of config,xshift=-3.35cm] (model) {\textbf{模型提交}\par 选择已适配结构，上传阶段权重\par 平台在固定测试输入上推理};

\node[wide,below=of score,fill=green!4] (report) {\textbf{查看与导出}\par 性能矩阵、BWTR、RMA、资源指标、客户端视图和任务样例};
